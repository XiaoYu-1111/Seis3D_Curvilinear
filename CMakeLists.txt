function(seis3d_target_options target)
    target_compile_definitions(${target} PRIVATE SEIS3D_OUTPUT_ROOT="${SEIS3D_OUTPUT_ROOT}")
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

function(seis3d_deploy_cuda target)
    if(SEIS3D_ENABLE_CUDA AND WIN32)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SEIS3D_CUDART_DLL}" "$<TARGET_FILE_DIR:${target}>" VERBATIM)
    endif()
endfunction()
